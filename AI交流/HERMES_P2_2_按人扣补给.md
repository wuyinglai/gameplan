### 3.2 `消耗补给.ts` —— 完整代码骨架

```typescript
// 🔴 src/p2/消耗补给.ts —— 🔴 纯计算，🔴 不碰时间/体力/天气
//
// 🔴 规则依据：3A §1.5（按出战人数）· §1.6（断粮断水后果）
// 🔴 🔴 本文件只做两件事：扣、罚。🔴 它不知道跨日这件事。

import type { 商队状态 } from './资源';

// ---------------------------------------------------------------------------
// 🔴 每天要扣多少（照 3A:89 的表）
// ---------------------------------------------------------------------------

/** 🔴 每人每天 1 份食物 + 1 份清水 —— 3A:65 用户亲口定 */
export function 每日消耗(出战人数: number): { 食物: number; 清水: number } {
  return { 食物: 出战人数, 清水: 出战人数 };
}

// ---------------------------------------------------------------------------
// 🔴 扣补给 —— 🔴 扣到 0 为止，🔴 不许扣成负数
// ---------------------------------------------------------------------------

export function 扣补给(状态: 商队状态, 人数: number): {
  扣了多少食物: number;
  扣了多少清水: number;
  食物耗尽: boolean;   // 🔴 扣完之后是 0 → 明天开始恶化
  清水耗尽: boolean;
} {
  const 要扣 = 每日消耗(人数);
  const 原食物 = 状态.资源.食物;
  const 原清水 = 状态.资源.清水;

  // 🔴 Math.max(0, ...) —— 🔴 不许出现负数（P1 的稳定性测试查过同类问题）
  状态.资源.食物 = Math.max(0, 原食物 - 要扣.食物);
  状态.资源.清水 = Math.max(0, 原清水 - 要扣.清水);

  return {
    扣了多少食物: 原食物 - 状态.资源.食物,
    扣了多少清水: 原清水 - 状态.资源.清水,
    食物耗尽: 状态.资源.食物 === 0,
    清水耗尽: 状态.资源.清水 === 0,
  };
}

// ---------------------------------------------------------------------------
// 🔴 断粮断水惩罚 —— 🔴 每天恶化一次
// ---------------------------------------------------------------------------
// 🔴 ⚠️ 角色系统还没有（P6 才做）—— 🔴 本步只留函数位，🔴 一律返回 0。
// 🔴 🔴 不许在这里自己造一个「虚拟血量」字段塞进商队状态 —— 那是 P6 的字段。

// 🔴 🔴 士气扣减写在 日结算02() 里，🔴 不在这里单独抽函数。
// 🔴 理由：断粮 + 缺水要**叠加成一个数**再扣，🔴 拆成两个函数反而要处理两次减法。

/** 🔴 断粮该扣多少血上限百分比 —— 10%（3A:105）。🔴 本步只返回数值，不作用在任何东西上。 */
export function 断粮扣血百分比(): number { return 10; }

/** 🔴 断水是断粮的两倍 —— 20%（3A:113）。 */
export function 断水扣血百分比(): number { return 20; }

// ---------------------------------------------------------------------------
// 🔴 描述 —— 给你自己看状态用
// ---------------------------------------------------------------------------

export function 描述补给(标签: string, 状态: 商队状态): string {
  return [
    `${标签}`,
    `  出战 ${状态.出战人数} 人 · 每天要 ${状态.出战人数} 份食物 + ${状态.出战人数} 份清水`,
    `  现有：食物 ${状态.资源.食物} 份 · 清水 ${状态.资源.清水} 份 · 士气 ${状态.资源.士气}`,
    `  ${状态.资源.食物 === 0 ? '🔴 断粮中 —— 明起每天恶化' : '✅ 有吃的'}`,
    `  ${状态.资源.清水 === 0 ? '🔴 缺水中 —— 明起每天恶化' : '✅ 有水'}`,
  ].join('\n');
}
```

### 3.3 `日结算02.ts` —— 完整代码骨架

```typescript
// 🔴 src/p2/消耗补给之外的第二个文件 —— 🔴 跨日那一刻调一次
//
// 🔴 🔴 本文件只做一件事：把扣补给 + 断粮惩罚串起来
// 🔴 🔴 它不许调 P1 的任何函数（阶段之间不许互相依赖）

import { 新建商队 } from './资源';
import { 扣补给, 描述补给 } from './消耗补给';
import type { 商队状态 } from './资源';

// ---------------------------------------------------------------------------
// 🔴 跨日那一刻跑一次
// ---------------------------------------------------------------------------

export interface 跨日结算结果 {
  扣了多少食物: number;
  扣了多少清水: number;
  /** 🔴 扣完之后是 0 —— 🔴 从**明天**开始恶化（今天已经扣过了） */
  食物耗尽: boolean;
  清水耗尽: boolean;
  /** 🔴 今天惩罚生效了吗 —— 🔴 判据 8 查的就是这个 */
  施加了惩罚: boolean;
  士气变化: number;
}

export function 日结算02(状态: 商队状态): 跨日结算结果 {
  // —— 第 1 步：扣 ——
  const 扣 = 扣补给(状态, 状态.出战人数);

  // —— 第 2 步：昨天就已经是 0 的，今天恶化一次 ——
  // 🔴 🔴 顺序：先扣，再罚。
  // 🔴 🔴 🔴 为什么不能反过来（这是我写指令时特意想清楚的一条）：
  // 🔴    如果「扣完为 0 就立刻罚」，那玩家开局带的粮正好等于一天消耗时，
  // 🔴    第一天就被罚了 —— 但 A 类说的是「**归零后，每天恶化一次**」，
  // 🔴    归零的那一刻不该罚，**从第二天开始才罚**。
  const 昨天食物就是零 = 状态.资源.食物 === 0 && 扣.扣了多少食物 === 0;
  const 昨天清水就是零 = 状态.资源.清水 === 0 && 扣.扣了多少清水 === 0;

  let 士气变化 = 0;
  if (昨天食物就是零 || 昨天清水就是零) {
    // 🔴 减 1 还是减 2？A 类说「食物和水都归零 → 双重惩罚叠加」
    // 🔴 🔴 这里**保守处理**：两者都归零才叠加成一个 -2，其余情况 -1。
    // 🔴 🔴 ⚠️ 这条 A 类没写清楚 —— 🔴 我按「叠加 = 两个 -1」处理，
    // 🔴    🔴 **报给我**，别自己改。
    士气变化 = (昨天食物就是零 ? 1 : 0) + (昨天清水就是零 ? 1 : 0);
    状态.资源.士气 = Math.max(0, 状态.资源.士气 - 士气变化);
  }

  return {
    扣了多少食物: 扣.扣了多少食物,
    扣了多少清水: 扣.扣了多少清水,
    食物耗尽: 扣.食物耗尽,
    清水耗尽: 扣.清水耗尽,
    施加了惩罚: 昨天食物就是零 || 昨天清水就是零,
    士气变化,
  };
}

/** 🔴 给你自己看结果用的 */
export function 跑一次看看(): void {
  const 状态 = 新建商队();
  状态.资源.食物 = 30;
  状态.资源.清水 = 30;
  console.log(描述补给('【跨日之前】', 状态));
  const r = 日结算02(状态);
  console.log(描述补给('【跨日之后】', 状态));
  console.log(`  扣了 ${r.扣了多少食物} 份食物 + ${r.扣了多少清水} 份清水 · 惩罚：${r.施加了惩罚 ? '有' : '无'}`);
}
```

### 8.1 🔴 这条链的顺序：先扣，再罚

🔴 **为什么不是「扣完为 0 就立刻罚」**：🔴 A 类写的是「**归零后，每天恶化一次**」。

🔴 如果玩家带的粮**正好等于一天消耗** —— 扣完变 0，🔴 如果立刻罚，**第一天就被罚了**。

🔴 但 A 类说的是「归零**后**」—— 🔴 **归零那一刻不算，从第二天算。**

### 8.2 🔴 断粮 + 缺水同时发生，士气减 2

🔴 **A 类只写了「双重惩罚叠加（90/90 → 70/70）」** —— 🔴 那个例子说的是**血量**。

🔴 **士气那块没写清是两个 −1 还是 −1。** 🔴 我按「叠加 = 两个 −1」处理。
