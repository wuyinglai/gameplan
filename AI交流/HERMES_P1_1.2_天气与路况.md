## 4. 🔴 两个文件的完整内容

### 4.1 `天气.ts`

```typescript
import type { 游戏状态, 天气名 } from './类型';

// ============================================================================
// 🔴 只负责：掷天气段 + 查每格加成。🔴 不管加成怎么用
// ============================================================================

/** 🔴 基础概率 —— 🔴 合计必须 = 1.00 */
export const 天气概率: Record<天气名, number> = {
  晴朗: 0.34, 阴天: 0.34,
  小雨: 0.05, 暴雨: 0.03,
  高温: 0.05, 酷热: 0.02,
  寒潮: 0.05, 严寒: 0.02,
  浓雾: 0.05, 重雾: 0.02,
  灰烬风: 0.02, 灰烬风暴: 0.01,
};

/** 🔴 段长范围 —— 含上下限 */
export const 天气段长: Record<天气名, [number, number]> = {
  晴朗: [5, 7],      阴天: [5, 7],
  小雨: [3, 6],      暴雨: [3, 6],
  高温: [3, 6],      酷热: [3, 6],
  寒潮: [3, 6],      严寒: [3, 6],
  浓雾: [3, 6],      重雾: [3, 6],
  灰烬风: [4, 5],    灰烬风暴: [2, 3],
};

/**
 * 🔴 掷一个天气段。
 * 🔴 随机源可注入 —— 🔴 测试要能指定，🔴 不许写死 Math.random
 */
export function 掷天气段(随机源: () => number = Math.random): 天气名 {
  const 点 = 随机源();
  let 累计 = 0;
  for (const 名 of Object.keys(天气概率) as 天气名[]) {
    累计 += 天气概率[名];
    if (点 < 累计) return 名;
  }
  return '灰烬风暴';   // 🔴 兜底（浮点误差时落到这里）
}

/**
 * 🔴 掷段长（整数，闭区间）
 */
export function 掷段长(天: 天气名, 随机源: () => number = Math.random): number {
  const [低, 高] = 天气段长[天];
  return 低 + Math.floor(随机源() * (高 - 低 + 1));
}

/**
 * 🔴 每格消耗加多少 —— 🔴🔴 数值待项目主裁定，先全 0
 * 🔴 裁定后只改这一个函数
 */
export function 每格加成(天: 天气名): number {
  // 🔴🔴 待裁定：1A §5.1 写 +5/+10，🔴 但与 5A 的恶劣清单矛盾
  return 0;
}

/**
 * 🔴 天气段推进（🔴 只允许 结算.跨日() 调用）
 * 🔴 剩余 −1；🔴 归零则重掷并写回 状态.当前天气 与 天气剩余天数
 */
export function 推进天气段(状态: 游戏状态, 随机源: () => number = Math.random): void {
  状态.天气剩余天数 -= 1;
  if (状态.天气剩余天数 <= 0) {
    状态.当前天气 = 掷天气段(随机源);
    状态.天气剩余天数 = 掷段长(状态.当前天气, 随机源);
  }
}
```

> 🔴 **🔴 关键**：`推进天气段()` **是唯一改 `当前天气` 与 `天气剩余天数` 的地方**。
> 🔴 **主循环不许重掷天气。**

### 4.2 `消耗.ts` 🔴 **每格消耗的唯一出口**

```typescript
import { 每格基准消耗 } from './体力';
import { 每格加成 as 天气加成 } from './天气';
import type { 路况档, 游戏状态 } from './类型';

// ============================================================================
// 🔴🔴🔴 全项目只有这一个函数算「走一格扣多少」
// 🔴 主循环只调它，🔴 天气模块不管、🔴 体力模块不管、🔴 地图模块不管
//
// 🔴 为什么必须唯一：一旦主循环自己也算一遍，
//   加天气时得同时改两个地方 → 漏改
//   加路况时又得改两个地方 → 重复扣
//   （旧代码踩过：加成被加了两次）
// ============================================================================

/** 🔴 路况加成表 —— 🔴🔴 待裁定，四档先全 0 */
export const 路况加成: Record<路况档, number> = { 0: 0, 1: 0, 2: 0, 3: 0 };

/** 🔴 公式就这么长 —— 不许加第五项，不许改顺序 */
export function 每格消耗(状态: 游戏状态, 目标格路况: 路况档): number {
  return 每格基准消耗 + 天气加成(状态.当前天气) + 路况加成[目标格路况];
}
```
