## 四、要做的事

### 1. 建两个文件
`src/b01/地图.ts`、`src/b01/状态.ts`

### 2. 把下面三段类型定义**原样抄成 TypeScript**，字段不许增删

#### （1）格子类型

> 🔴 **山/水/桥/林/废墟必须是顶层枚举值，不能折进一个叫"地形"的笼统值。**
> 🔴 （原因：之前折进去之后，`不可通行类型.has(格.类型)` 恒等于 false —— 山和水谁也拦不住）

```typescript
export type 格子类型 =
  | '空地'
  | '山' | '水'                 // 🔴 不可通行
  | '桥' | '林' | '废墟'          // 🔴 可通行（废墟本轮按可通行处理）
  | '商路'
  | '大城' | '小城' | '城镇' | '驿站' | '村庄' | '营地';

export type 路况档 = '好' | '普通' | '差' | '恶劣';

export interface 格子 {
  坐标: { x: number; y: number };
  类型: 格子类型;
  路况档: 路况档;
  地点名: string | null;
  已到访: boolean;      // 🔴 记的是"玩家实际到没到过"，不是"能不能看见"
  信息等级: number;      // 🔴 本轮没人读它（主动侦察归单元04），只是随格存着
}

// 🔴 这两个集合里的值必须都在上面的 格子类型 枚举里
//    写了枚举外的值 = 这段代码永远不会生效（TypeScript 会报错，那正是我们要的）
export const 不可通行类型: ReadonlySet<格子类型> = new Set<格子类型>(['山', '水']);
export const 可停留类型: ReadonlySet<格子类型> = new Set<格子类型>(
  ['大城', '小城', '城镇', '驿站', '村庄', '营地']
);
```

#### （2）游戏状态

```typescript
export type 天气类型 =
  | '晴朗' | '阴天' | '小雨' | '暴雨'
  | '高温' | '酷热' | '寒潮' | '严寒'
  | '浓雾' | '重雾' | '灰烬风' | '灰烬风暴';

export interface 游戏状态 {
  当前坐标: { x: number; y: number };
  地图: Map<string, 格子>;        // key = `${x},${y}`

  第几天: number;                 // 从 1 开始
  当前小时: number;               // 7 ~ 19（19:00 即收工）
  // 🔴 这两个字段只有 结算.ts 的 跨日() 能改，其他模块只读

  野外体力: number;               // 上限 100
  体力上限: number;               // 恒为 100

  当前天气: 天气类型;             // 🔴 存具体天气名，不存布尔
  天气剩余天数: number;

  已到访地点: string[];           // 权威字段是 格子.已到访，这个列表只是它的投影
}
```
