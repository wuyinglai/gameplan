### 3.1 `消耗补给.ts`

```typescript
// 🔴 src/p2/消耗补给.ts —— 2B 中阶段 2.1
//
// 🔴 规则依据：3A:65 / 3A:85 / 3A:89 / 3A:93 / 3A:97 / 3A:105 / 3A:113
// 🔴 本文件只做「算要多少」+「扣掉」+「断粮惩罚」三件事。
// 🔴 🔴 它不碰跨日 · 不碰血量（角色系统在 6B）· 不知道有下一条链存在。

import type { 商队状态 } from './资源';

// ============================================================================
// 🔴 要扣多少
// ============================================================================

/**
 * 每人每天 1 份食物 + 1 份清水。
 * 🔴 出处：3A:65（2026-10-13 用户亲口定）· 3A:89 的表
 *
 * 🔴🔴 返回值必须是两个字段，不许合并成一个总数。
 * 🔴 出处：3A:93「🔴 表述铁律：两种资源永远分开写，🔴 **禁止合并成「28 份」这类总数**」
 */
export function 每日消耗(出战人数: number): { 食物: number; 清水: number } {
  return { 食物: 出战人数, 清水: 出战人数 };
}

// ============================================================================
// 🔴 扣掉
// ============================================================================

export interface 扣补给结果 {
  /** 🔴 实际扣掉几份 —— 可能比要扣的少（不够的时候） */
  扣了多少食物: number;
  扣了多少清水: number;
  /** 🔴 扣完之后是 0 —— 🔴 本文件**只报告**，🔴 惩罚在下一段 */
  食物耗尽: boolean;
  清水耗尽: boolean;
}

/**
 * 从状态里扣掉今天要吃的。
 *
 * 🔴🔴 扣到 0 为止，**不许扣成负数** —— Math.max(0, ...)。
 * 🔴 🔴 本函数**不碰士气、不碰血量** —— 那是下一段的事。
 */
export function 扣补给(状态: 商队状态): 扣补给结果 {
  const 要扣 = 每日消耗(状态.出战人数);
  const 原食物 = 状态.资源.食物;
  const 原清水 = 状态.资源.清水;

  状态.资源.食物 = Math.max(0, 原食物 - 要扣.食物);
  状态.资源.清水 = Math.max(0, 原清水 - 要扣.清水);

  return {
    扣了多少食物: 原食物 - 状态.资源.食物,
    扣了多少清水: 原清水 - 状态.资源.清水,
    食物耗尽: 状态.资源.食物 === 0,
    清水耗尽: 状态.资源.清水 === 0,
  };
}

// ============================================================================
// 🔴 断粮断水惩罚
// ============================================================================

/** 🔴 断粮扣多少血上限百分比 —— 10%（3A:105）。🔴 本阶段只返回数值，不作用在任何东西上。 */
export function 断粮扣血百分比(): number { return 10; }

/** 🔴 断水是断粮的两倍 —— 20%（3A:113）。 */
export function 断水扣血百分比(): number { return 20; }

/**
 * 🔴 断粮断水 → 扣士气。
 *
 * 🔴 A 类：食物 = 0 → 士气 −1；清水 = 0 → 士气 −1（3A:105 / 3A:113）
 * 🔴 A 类：两者都归零 → **双重惩罚叠加**（3A:129）
 *
 * 🔴🔴 士气下限 0 —— 🔴 不许扣成负数。
 * 🔴🔴 血上限 / 当前血 / 重伤恶化 🔴 本阶段**不做**（角色系统在 6B），
 *    🔴 上面两个「扣血百分比」函数只是把 A 类的数字**记下来**，🔴 不作用。
 */
export function 断粮扣士气(状态: 商队状态, 食物耗尽: boolean, 清水耗尽: boolean): number {
  let 扣 = 0;
  if (食物耗尽) 扣 += 1;
  if (清水耗尽) 扣 += 1;
  const 前 = 状态.资源.士气;
  状态.资源.士气 = Math.max(0, 前 - 扣);
  return 前 - 状态.资源.士气;
}

// ============================================================================
// 🔴 描述 —— 给你自己看状态用
// ============================================================================

export function 描述补给(标签: string, 状态: 商队状态): string {
  const 要 = 每日消耗(状态.出战人数);
  return [
    `${标签}`,
    `  出战 ${状态.出战人数} 人 · 每天要 ${要.食物} 份食物 + ${要.清水} 份清水`,
    `  现有：食物 ${状态.资源.食物} 份 · 清水 ${状态.资源.清水} 份 · 士气 ${状态.资源.士气}`,
    `  ${状态.资源.食物 === 0 ? '🔴 食物 = 0' : '✅ 有食物'}`,
    `  ${状态.资源.清水 === 0 ? '🔴 清水 = 0' : '✅ 有清水'}`,
  ].join('\n');
}
```

### 3.2 `日结算02.ts`

```typescript
// 🔴 src/p2/日结算02.ts —— 2B 中阶段 2.1
//
// 🔴 🔴 2B 的跨日链。🔴 它跟 1B 的 结算.跨日() 是**两条并行的链**。
// 🔴 🔴 本文件**不许 import src/p1/** —— 阶段之间不许互相依赖。
// 🔴 🔴 本文件**不负责被谁调用** —— 调用方是 2B 最后一个中阶段的事。

import { 新建商队 } from './资源';
import { 扣补给, 断粮扣士气, 描述补给 } from './消耗补给';
import type { 商队状态 } from './资源';

// ============================================================================
// 🔴 跨日那一刻跑一次
// ============================================================================

export interface 跨日结算结果 {
  扣了多少食物: number;
  扣了多少清水: number;
  /** 🔴 扣完之后是 0 —— 🔴 从**明天**开始恶化（今天已经扣过了） */
  食物耗尽: boolean;
  清水耗尽: boolean;
  /** 🔴 今天惩罚生效了吗 */
  施加了惩罚: boolean;
  士气变化: number;
}

/**
 * 2B 的跨日结算 —— **一天只调一次**。
 *
 * 🔴🔴 顺序是「先扣，再罚」，而且「归零那一刻不罚」：
 * 🔴   A 类原文是「食物/水 = 0 后，**每天恶化一次**」（3A:105）
 * 🔴   玩家带的粮正好等于一天消耗时，扣完变 0 —— 🔴 **归零那一刻不罚**，
 * 🔴   **从第二天开始才罚**。🔴 否则玩家第一天就被罚了。
 */
export function 日结算02(状态: 商队状态): 跨日结算结果 {
  // —— 第 1 步：扣 ——
  const 扣 = 扣补给(状态);

  // —— 第 2 步：昨天就已经是 0 的，今天恶化一次 ——
  // 🔴 「扣了多少 = 0」= 说明扣之前就是 0（不够扣）
  const 昨天食物就是零 = 状态.资源.食物 === 0 && 扣.扣了多少食物 === 0;
  const 昨天清水就是零 = 状态.资源.清水 === 0 && 扣.扣了多少清水 === 0;

  const 施加了惩罚 = 昨天食物就是零 || 昨天清水就是零;
  const 士气变化 = 施加了惩罚
    ? 断粮扣士气(状态, 昨天食物就是零, 昨天清水就是零)
    : 0;

  return {
    扣了多少食物: 扣.扣了多少食物,
    扣了多少清水: 扣.扣了多少清水,
    食物耗尽: 扣.食物耗尽,
    清水耗尽: 扣.清水耗尽,
    施加了惩罚,
    士气变化,
  };
}

/** 🔴 给你自己看结果用的 */
export function 跑一次看看(): void {
  const 状态 = 新建商队();
  状态.资源.食物 = 30;
  状态.资源.清水 = 30;
  console.log(描述补给('【跨日之前】', 状态));
  const r = 日结算02(状态);
  console.log(描述补给('【跨日之后】', 状态));
  console.log(`  扣了 ${r.扣了多少食物} 份食物 + ${r.扣了多少清水} 份清水 · 惩罚：${r.施加了惩罚 ? '有' : '无'}`);
}
```

### 8.1 🔴 顺序：先扣，再罚，且「归零那一刻不罚」
