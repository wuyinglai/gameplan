## 3. 🔴 四个文件的完整内容

### 3.1 `时间.ts`

```typescript
import type { 游戏状态 } from './类型';

// ============================================================================
// 🔴 只负责：推进动作耗时 + 报告是否到日界
// 🔴 🔴 不许改 第几天、不许改 当前小时 —— 那是 结算.ts 的事
// ============================================================================

export const 每格耗时 = 2;
export const 开工小时 = 7;
export const 收工小时 = 19;

/** 🔴 推进 n 小时（🔴 只加数，不碰日期）。返回 true = 到点收工了 */
export function 推进(状态: 游戏状态, n: number): boolean {
  状态.当前小时 += n;
  return 状态.当前小时 >= 收工小时;
}

/** 🔴 当前小时还能不能再走一格。🔴 17:00 还能走（17+2=19 正好收工） */
export function 时间够吗(状态: 游戏状态): boolean {
  return 状态.当前小时 + 每格耗时 <= 收工小时;
}
```

> 🔴 **🔴 关键**：`推进()` **只加小时数**，🔴 **它不许碰 `第几天`**。
> 🔴 🔴 **一旦它把日期也改了，一天就会多跨一次** —— 🔴 旧代码踩过。

### 3.2 `体力.ts`

```typescript
export const 体力上限值 = 100;
export const 每日回复 = 50;

// ============================================================================
// 🔴 只负责：够不够、扣、回。🔴 不管扣完触发什么
// ============================================================================

/** 🔴 够不够。🔴 判断必须在这里，不许散在主循环里 */
export function 够吗(状态: 游戏状态, 要扣: number): boolean {
  return 状态.野外体力 >= 要扣;
}

/** 🔴 扣。🔴 只允许在「已确认够」之后调用 —— 调用方负责先问 */
export function 扣(状态: 游戏状态, 多少: number): void {
  状态.野外体力 -= 多少;
}

/** 🔴 每日回复。🔴 🔴 只允许 结算.跨日() 调用。封顶 100 */
export function 回(状态: 游戏状态): void {
  状态.野外体力 = Math.min(体力上限值, 状态.野外体力 + 每日回复);
}
```

> 🔴 **🔴 关键**：`回()` **只允许被 `跨日()` 调一次**。
> 🔴 🔴 **主循环和结算各回一次 → 一天回两份** —— 🔴 旧代码实测过该拿 110 实际拿了 100。

### 3.3 `结算.ts` 🔴 **跨日的唯一入口**

```typescript
import { 回 } from './体力';
import { 推进天气段 } from './天气';
import { 开工小时 } from './时间';
import type { 游戏状态 } from './类型';

// ============================================================================
// 🔴🔴🔴 全项目只有这一处能改 第几天 和 当前小时
// 🔴 任何别的地方碰这两个字段都是 bug
// ============================================================================

/**
 * 🔴 跨日 —— 一次做四件事，顺序固定，不许增删
 *   1. 第几天 += 1
 *   2. 当前小时 = 07:00
 *   3. 体力 += 50（封顶 100）
 *   4. 天气段 −1（到期则重掷）
 */
export function 跨日(状态: 游戏状态): void {
  状态.今天已结算 = true;
  状态.第几天 += 1;
  状态.当前小时 = 开工小时;
  回(状态);
  推进天气段(状态);
}
```

> ⚠️ **天气那条先接一个空函数**，🔴 第 3 步才实现它 —— 🔴 但**调用位置现在就要留好**，
> 🔴 不许等第 3 步再改 `跨日()` 的结构。

### 3.4 `结束当天.ts`

```typescript
import { 跨日 } from './结算';
import type { 操作结果, 游戏状态 } from './类型';

/**
 * 🔴 「今天到此为止」
 * 🔴 任何时候都能用 —— 体力 0 点、15:00 都能用（🔴 否则玩家会卡死）
 * 🔴 只做一件事：调一次 跨日()
 * 🔴 🔴 不许：不扣移动体力 · 不推进 2 小时 · 不改坐标 · 不判断天气
 */
export function 结束当天(状态: 游戏状态): 操作结果 {
  if (状态.今天已结算) {
    return { 成功: false, 原因: '今天已结算过了' };
  }
  跨日(状态);
  return { 成功: true, 原因: null };
}
```
