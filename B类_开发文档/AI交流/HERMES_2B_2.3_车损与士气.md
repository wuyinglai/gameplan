### 3.2 `士气.ts`

```typescript
// 🔴 src/p2/士气.ts —— 2B 中阶段 2.3
// 规则依据：3A:499-513（消费）· 3A:515-532（自然恢复）· 3A:495-497（封顶）
import type { 商队状态 } from './资源';
import type { 地点类型 } from './资源';

export const 士气上限常量 = 10; // 🔴 3A:512（资源.ts 已有 士气上限=10，此处只引用口径，不重复定义数值源）

/** 🔴 3A:503-507 消费三档（⚠️ 参考值 · 判据锁死 · 调平只改这张表） */
export const 士气消费表 = [
  { 名字: '酒馆小聚', 价格: 150, 恢复: 1, 生效条件: 6 },   // 士气 6 以下有效
  { 名字: '商队宴席', 价格: 400, 恢复: 1, 生效条件: 10 },  // 10 以下有效
  { 名字: '豪华休整', 价格: 800, 恢复: 2, 生效条件: null }, // 不限
] as const;

/** 🔴 3A:509-513：花钱恢复 —— 可多次 · 封顶 10 · 🔴 绕开自然恢复的上限与冷却 */
export function 士气消费(状态: 商队状态, 档名: '酒馆小聚' | '商队宴席' | '豪华休整'): { 成功: boolean; 原因: string | null } {
  const 档 = 士气消费表.find((x) => x.名字 === 档名)!;
  if (状态.资源.士气 >= 士气上限常量) return { 成功: false, 原因: '士气已到上限' };       // 3A:512
  if (档.生效条件 !== null && 状态.资源.士气 >= 档.生效条件) return { 成功: false, 原因: `${档名}只在士气 ${档.生效条件} 以下有效` };
  if (状态.资源.金钱 < 档.价格) return { 成功: false, 原因: '钱不够' };
  状态.资源.金钱 -= 档.价格;
  状态.资源.士气 = Math.min(士气上限常量, 状态.资源.士气 + 档.恢复);
  return { 成功: true, 原因: null };
}

/** 🔴 3A:517-525 自然恢复：地点 +1 · 城市≤6 / 城镇驿站≤4 / 村庄野外 0 · 同一地点 20 天冷却 */
export function 自然恢复上限(地点: 地点类型): number {
  return { 大城: 6, 小城: 6, 城镇: 4, 驿站: 4, 村庄: 0 }[地点]; // 🔴 3A:519-524
}
export const 自然恢复冷却天数 = 20; // 🔴 3A:517

/**
 * 抵达地点时的自然恢复。冷却表由调用方传入（今日 · 地点 → 上次恢复日），
 * 本函数不改冷却表本身（2.4 串联时由跨日链统一记账）。
 */
export function 自然恢复(
  状态: 商队状态, 地点: 地点类型, 今日: number, 冷却表: Map<string, number>,
): { 恢复了: number; 原因: string } {
  const 上限 = 自然恢复上限(地点);
  if (上限 === 0) return { 恢复了: 0, 原因: `${地点}不恢复士气` };              // 3A:524-525
  if (状态.资源.士气 >= 上限) return { 恢复了: 0, 原因: `士气 ${上限} 以上不自然恢复` };
  const 键 = `${地点}`;
  const 上次 = 冷却表.get(键);
  if (上次 !== undefined && 今日 - 上次 < 自然恢复冷却天数) {
    return { 恢复了: 0, 原因: `${地点} 20 天冷却中` };                            // 3A:517
  }
  状态.资源.士气 = Math.min(士气上限常量, 状态.资源.士气 + 1);
  return { 恢复了: 1, 原因: 'ok' };
}

/** 🔴 3A:495-497 扣减封顶：同场战斗 ≤3 · 同一天 ≤3（崩溃撤退不占）—— 2.1 的断粮扣不走这里，不许改 */
export function 当日士气扣减上限(状态: 商队状态, 拟扣: number, 当日已扣: number): number {
  void 状态;
  const 场内 = Math.min(拟扣, 3);
  return Math.min(场内, Math.max(0, 3 - 当日已扣));
}
```
