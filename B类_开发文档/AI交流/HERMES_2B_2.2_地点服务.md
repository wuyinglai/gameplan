### 1.4 🔴 库存限制（`2A §7.2`）

🔴 **A 类给了库存区间**（🔴 我第一次写指令时说「没给」是错的，🔴 这里更正）：

| 地点 | 补给库存 | 零件库存 | 药包库存 |
|:--|:--|:--|:--|
| 🔴 **大城 / 小城** | **大量 / 近似不限** | **大量 / 近似不限** | **大量 / 近似不限** |
| 🔴 **城镇** | **20–40** | **3–6** | **5–10** |
| 🔴 **驿站** | **15–30** | **5–8** | **3–6** |

🔴 **本步怎么处理（🔴 我的判断，🔴 你可以不同意）**：
🔴 **用区间下限当「单次可买上限」** —— 🔴 城镇食物最多买 20 份 · 驿站最多 15 份。

🔴 **理由**：🔴 区间是「这一批有多少」，🔴 **不是「一次能买多少」** —— 🔴 A 类没说。
🔴 **取下限最保守**（🔴 不会让玩家买超库存），🔴 **而且下限是确定值好测**。
🔴 **上限取太大** → 玩家能买空整个城镇；🔴 **上限取 0** → 城镇驿站完全不能补给，🔴 **跟「补给容错点」的定位矛盾**。

🔴 **村庄（🔴 2026-10-24 项目主裁定）** —— 🔴 **能买，但量少价高**：食物单价 **20 金/份** · 单次上限 **10 份**（= 2 格）。
🔴 **村庄清水没提** → 填 0 = 本步不卖（🔴 剩余待裁定⑲ 只剩清水一项）。零件/药包村庄不卖（`2A §6`）。

### 3.2 `地点服务.ts`

```typescript
// 🔴 src/p2/地点服务.ts —— 2B 中阶段 2.2
//
// 🔴 规则依据：2A §7.1（价格表）· 2A §2~§6（各地点能做什么）· 3A §2.4（修理规则）· 3A:35-36（格换算）
// 🔴 本文件只做两件事：买（食物/清水/药包/零件）· 修车
// 🔴 🔴 它不碰治疗 / 招募 / 士气消费 / 买卖货物 / 接单 / 跨日链

import type { 商队状态, 地点类型 } from './资源';

// ============================================================================
// 🔴 价格表 —— 照抄 2A §7.1
// ============================================================================

/** 🔴 买 1 份的价格（金）。🔴 0 = 🔴 A 类还没定 / 这地方不卖 */
export const 价格: Record<地点类型, {
  食物: number; 清水: number; 零件: number; 药包: number;
}> = {
  // 🔴 大城 = 物价最低（2A:133）· 🔴 基础补给 10 金/份是不可动的锚
  大城: { 食物: 10, 清水: 8, 零件: 50, 药包: 50 },
  // 🔴 小城 = 与大城基本相同（2A:141）
  小城: { 食物: 10, 清水: 8, 零件: 50, 药包: 50 },
  // 🔴 城镇 = 生活物资与药包便宜，零件修车贵（2A:76）
  城镇: { 食物: 15, 清水: 0, 零件: 75, 药包: 70 },
  // 🔴 驿站 = 车队机械件便宜（零件/修车），生活物资贵（2A:77）
  驿站: { 食物: 16, 清水: 0, 零件: 70, 药包: 75 },
  // 🔴 村庄 = 量少价高（2026-10-24 项目主裁定 + 废案 Demo数值手册 §4.1）：食物 20 金/份
  村庄: { 食物: 20, 清水: 0, 零件: 0, 药包: 0 },
};

/** 🔴 格换算（3A:35-36）· 🔴 零件/药包 A 类没写 → 暂按 1 个 = 1 格 */
export const 每格份数: Record<'食物' | '清水' | '零件' | '药包', number> = {
  食物: 5, 清水: 3, 零件: 1, 药包: 1,
};

// ============================================================================
// 🔴 买
// ============================================================================

export interface 买的结果 {
  成功: boolean;
  原因: string | null;
  /** 🔴 实际买到的份数 / 个数 */
  买到食物: number;
  买到清水: number;
  买到零件: number;
  买到药包: number;
  /** 🔴 实际花了多少金 */
  花了多少金: number;
}

/** 🔴 单次能买多少（🔴 库存区间的下限 —— 见指令 §1.4）· 🔴 null = 近似不限 · 0 = 买不到 */
export const 库存上限: Record<地点类型, {
  食物: number | null; 清水: number | null; 零件: number | null; 药包: number | null;
}> = {
  大城: { 食物: null, 清水: null, 零件: null, 药包: null },  // 🔴 null = 近似不限
  小城: { 食物: null, 清水: null, 零件: null, 药包: null },
  城镇: { 食物: 20, 清水: 0, 零件: 3, 药包: 5 },
  驿站: { 食物: 15, 清水: 0, 零件: 5, 药包: 3 },
  村庄: { 食物: 10, 清水: 0, 零件: 0, 药包: 0 },        // 🔴 村庄食物单次上限 10 份（🔴 2026-10-24 项目主裁定「约 10 份」）
};

/**
 * 🔴 在某个地点买补给。
 *
 * 🔴🔴 一次只买一种（🔴 简单好测）· 🔴 要买几种就调几次。
 *
 * @param 种类 '食物' | '清水' | '零件' | '药包'
 * @param 格数 买几格（🔴 按格买，不拆散份 —— 3A:60）
 */
export function 买补给(
  状态: 商队状态, 地点: 地点类型, 种类: '食物' | '清水' | '零件' | '药包', 格数: number,
): 买的结果 {
  const 空 = { 成功: false, 原因: '', 买到食物: 0, 买到清水: 0, 买到零件: 0, 买到药包: 0, 花了多少金: 0 };

  if (格数 <= 0) return { ...空, 原因: '格数不对' };

  const 单价 = 价格[地点][种类];
  if (单价 === 0) return { ...空, 原因: `${地点}不卖${种类}` };

  const 每格 = 每格份数[种类];
  const 总份数 = 格数 * 每格;
  // 🔴 单价是「每份/每个」价（2A:199 锚：基础补给 10 金/份）→ 总价 = 单价 × 份数（🔴 不是 × 格数）
  const 总价 = 单价 * 总份数;

  if (状态.资源.金钱 < 总价) return { ...空, 原因: '钱不够' };

  // 🔴 库存检查 —— 🔴 null = 近似不限（城市）· 🔴 0 = 买不到（任何正数份数都超 0）
  const 上限 = 库存上限[地点][种类];
  if (上限 !== null && 总份数 > 上限) return { ...空, 原因: '库存不够' };

  // 🔴 扣钱 + 进货物（🔴 四种分开记，不合并）
  状态.资源.金钱 -= 总价;
  if (种类 === '食物') 状态.资源.食物 += 总份数;
  else if (种类 === '清水') 状态.资源.清水 += 总份数;
  else if (种类 === '零件') 状态.资源.零件 += 总份数;
  else 状态.资源.药包 += 总份数;

  const 结果: 买的结果 = {
    成功: true, 原因: null,
    买到食物: 0, 买到清水: 0, 买到零件: 0, 买到药包: 0,
    花了多少金: 总价,
  };
  if (种类 === '食物') 结果.买到食物 = 总份数;
  else if (种类 === '清水') 结果.买到清水 = 总份数;
  else if (种类 === '零件') 结果.买到零件 = 总份数;
  else 结果.买到药包 = 总份数;
  return 结果;
}

// 🔴 修车
// ============================================================================

/** 🔴 花钱修车要多少金（3A:319-321）· 🔴 0 = 这地方不能花钱修 */
export const 修理价格: Record<地点类型, number> = {
  大城: 10, 小城: 10, 城镇: 30, 驿站: 25, 村庄: 0,
};

/** 🔴 花钱修一次恢复多少耐久（3A:319-321）—— 🔴 全部是 20 */
export const 修理恢复 = 20;

/** 🔴 用零件修一次恢复多少耐久（3A:322）—— 普通（无修补师）那一档 */
export const 零件修复耐久 = 10;

export interface 修车结果 {
  成功: boolean;
  原因: string | null;
  花了多少金: number;
  用了几个零件: number;
  恢复了多少耐久: number;
}

/**
 * 🔴 修车。两种方式：花钱（按地点价格）· 用零件（1 个 = 10 耐久）。
 *
 * 🔴🔴 耐久上限是多少 —— 🔴 **A 类没定**（`资源.ts` 里 `车队耐久上限 = 0`），
 * 🔴 所以本步**不许自己定一个上限**。🔴 恢复的量算出来，🔴 **上限由调用方保证**。
 */
export function 修车(
  状态: 商队状态, 地点: 地点类型, 方式: '花钱' | '用零件',
): 修车结果 {
  const 空 = { 成功: false, 原因: '', 花了多少金: 0, 用了几个零件: 0, 恢复了多少耐久: 0 };

  if (方式 === '花钱') {
    const 单价 = 修理价格[地点];
    if (单价 === 0) return { ...空, 原因: `${地点}不能花钱修车` };
    if (状态.资源.金钱 < 单价) return { ...空, 原因: '钱不够' };
    状态.资源.金钱 -= 单价;
    状态.资源.耐久 += 修理恢复;
    return { 成功: true, 原因: null, 花了多少金: 单价, 用了几个零件: 0, 恢复了多少耐久: 修理恢复 };
  }

  // 用零件
  if (状态.资源.零件 < 1) return { ...空, 原因: '零件不够' };
  状态.资源.零件 -= 1;
  状态.资源.耐久 += 零件修复耐久;
  return { 成功: true, 原因: null, 花了多少金: 0, 用了几个零件: 1, 恢复了多少耐久: 零件修复耐久 };
}
```

## 8. 🔴 我在这条指令里做过的判断（🔴 你可以不同意，但要告诉我）

### 8.1 🔴 「一次只买一种」

🔴 **为什么不分一次买齐**：🔴 一次买齐要处理「钱够买 A 但不够买 B」的情况，
🔴 🔴 **规则没定这种部分成交怎么办**。🔴 **一次买一种最简单也最不容易错。**

🔴 **要买几种就调几次** —— 🔴 那是调用方的事。

### 8.2 🔴 城镇 / 驿站的清水 → 填 0 = 当作「不卖」

🔴 **A 类原文**（`2A:199`）：「🔴 **城镇 / 驿站的清水价尚未定**，待 `23` / `24` 按食物的同档比例（×1.5 / ×1.6）推导后回填，🔴 **本表不手写**」。

🔴 **本步处理**：🔴 **价格填 0** → 🔴 **代码里当作「不卖」** → 🔴 **判据 11 就是测这个**。

🔴 🔴 **这一条是我定的** —— 🔴 **另一种处理是「随便填个数先跑起来」**，🔴 我不建议（🔴 **编出来的价格会锁进测试**）。

### 8.3 🔴 零件 / 药包 → 1 个 = 1 格

🔴 **A 类只写了食物 1 格 5 份、清水 1 格 3 份**，🔴 **零件和药包没写几格一个**。

🔴 **本步按「1 个 = 1 格」处理** —— 🔴 🔴 **这是我定的**，🔴 **报给我**。

### 8.4 🔴 修车不设上限

🔴 **A 类里耐久上限是 0（未定）** → 🔴 **本步不许自己定一个数字**。
🔴 **恢复量算出来，🔴 上限由调用方保证**（2B 后面）。
