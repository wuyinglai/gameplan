  | '村庄'

  类型: 格子类型;

  /** 🔴 不可通行 = 山 / 水 / 裂谷。地形里桥和碎石地是可通行的 */
  可通行: boolean;
  /** 🔴 走进地点类格子后置 true —— 用于「已到访」 */
  已到访: boolean;
  /** 🔴 地点名（大城名等）—— 非地点格为 null。P1 不用，P2~P7 用 */
  地点名: string | null;
}

/** 🔴 地图 json 的 t 键 → 格子类型。🔴 唯一翻译处，不许在别处再判一次 */
export const T到类型: Record<string, 格子类型> = {
  big: '大城',
  small: '小城',
  town: '城镇',
  village: '村庄',
  inn1: '驿站',
  // 🔴 已知偏差：json 写 special，规格书写「探索点」，A 类 §4 没有这一类
  //    暂时映射成「营地」（唯一功能相近的临时落脚点）—— 见 §6.4
  special: '营地',
  road: '空地',        // 🔴 商路不是格子类型，是「路况好」的空地
  blk_fr: '地形',      // 可通行（桥 / 碎石地）
  block: '地形',       // 不可通行（山 / 裂谷）
  blk_wt: '地形',      // 不可通行（水 / 河）
};

/** 🔴 不可通行的 t 键 —— 只有这三个。其余全部可通行 */
export const 不可通行的T: ReadonlySet<string> = new Set(['block', 'blk_wt']);

/** 🔴 地点类格子 —— 踩上去 = 进界面，不是「继续赶路」 */
export const 地点类: ReadonlySet<格子类型> = new Set<格子类型>([
  '大城', '小城', '城镇', '驿站', '村庄', '营地',
]);

/** 🔴 不可通行的格子类型 —— 只有「地形」里的一部分，由 t 键决定 */
export function 该T是否可通行(t: string): boolean {
  return !不可通行的T.has(t);
}

## §4 🔴 主循环：走一格的九步（顺序一步不许调换）

```
玩家输入"走一格" → 目标坐标
   ↓
① 目标坐标在地图内吗？    0 ≤ x < 400 且 0 ≤ y < 400
   └── 否 ──→ 拒绝「出地图了」🔴 什么都不扣 ──→ 回到 ①
   ↓ 是
② 目标格与当前格四方向相邻吗？   |dx| + |dy| == 1
   └── 否 ──→ 拒绝「不是相邻格」（斜向 / 隔两格 / 原地都走不了）
   ↓ 是
③ 目标格可通行吗？        查格(x, y).可通行
   └── 否 ──→ 拒绝「过不去」
   ↓ 是   🔴 到这里手上已经有目标格 = 天气和路况都能读了
④ 体力够吗？             够吗(状态, 每格消耗(状态, 目标格.路况))
   └── 否 ──→ 拒绝「体力不够」
   ↓ 够
⑤ 时间够走完这一格吗？   当前小时 + 2 ≤ 19
   └── 否 ──→ 拒绝「时间不够」
   ↓ 够
⑥ 🔴 全部检查通过 —— 现在才允许改状态：
     体力   −= 每格消耗(状态, 目标格.路况)    ← 🔴 唯一出口，不自己算
     当前坐标 = 目标格
     推进(状态, 2)                            ← 只推时间，不跨日
   ↓
⑦ 推进后到 19:00 了吗？
   └── 没有 ──→ 标记该格 已到访（如果是地点类）→ 回到 ①
   │ 到了
   ↓
⑧ 🔴 调 跨日(状态) —— 🔴 只调这一次
   ↓
⑨ 走进的是没到访过的地点吗？ 是 → 置 已到访 = true
   ↓
   走到终点了吗？ 是 → 返回「已抵达」（P1 到此为止，🔴 进城买东西是 P2）
   ↓
   回到 ①
```

### 🔴 三条不能破的规则

| # | 规则 | 违反后果 |
|:-:|:--|:--|
| 1 | 🔴 **①②③④⑤ 任何一条不过 → 体力、时间、坐标、天气全部不变** | 玩家点一下扣一次血 |
| 2 | 🔴 **⑧ 只调一次** —— 主循环不许自己回体力 | 一天回两份体力（+100 而不是 +50） |
