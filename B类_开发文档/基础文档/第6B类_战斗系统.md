## 3. 数据结构（6.0 本批全部落地）

```typescript
// —— 单位（敌我同构 · 对等原则 4A:287）——
export interface 状态实例 { 名字: string; 层数: number }        // 具体效果 6.3 起
export interface 单位 {
  名字: string; 阵营: '我方' | '敌方';
  当前生命: number; 生命上限: number;
  护甲: number;      // ✅ 4A:273 开战赋值·战内只减不增·归零转无甲（档判定在 6.2）
  护盾: number;      // ✅ 4A:170 开战 0·回合末清空·可给任意队友
  已退出: boolean;   // ✅ 4A:641 生命归零→退出当前战斗
  本场重伤次数: number;  // ✅ 4A:643-646 归零一次 +1（四级定级在 战后.ts）
  状态: 状态实例[];
}
// —— 体力池 ✅ 4A:78-92 ——
export interface 体力池 { 当前: number; 上限: number; 每回合回复: number }  // 上限=2×回复 · 起手=上限
// —— 战斗状态机 ——
export interface 战斗状态 {
  回合: number;            // 从 1 起
  难度: 战斗难度;
  我方: 单位[]; 敌方: 单位[];
  体力池: 体力池;           // 全队共享（我方专用；敌方行动不花这个池）
  药包剩余: number; 零件剩余: number; 商队耐久: number;
  撤退尝试次数: number;
  结局: null | '胜利' | '失败' | '撤退成功';   // 6.0 只判 胜/败；撤退 6.5
  随机种子: number;         // 🟡 判定用可注入种子（撤退/迟滞/敌方AI），保证测试可复现
}
```

## 4. 函数签名（6.0 本批）

```typescript
export function 开始战斗(入参: 战斗入参): 战斗状态
//   深拷贝快照→内部单位；体力池：Σ个人体力（快照缺省 50）→ 回复=Σ 上限=2Σ 起手=上限
export function 判定结局(状态: 战斗状态): '胜利' | '失败' | null
//   敌方全部已退出→胜利；我方全部已退出→失败；否则 null（✅ 4A:641 归零=退出）
export function 进入下一回合(状态: 战斗状态): void
//   回合+1 · 体力池+回复（封顶上限）· 双方护盾清零（✅ 4A:170/281）· 状态tick 6.3 接
export function 单位归零(状态: 战斗状态, 单位: 单位): void
//   已退出=true · 本场重伤次数+1（敌方死亡→逃跑意愿钩子 6.4 接）
```

## 5. 战斗主循环（6.1 起逐步成型，6.0 先立骨架）

```
开始战斗(入参) → 状态
循环 {
  我方回合：抽3张 → 出牌（体力闸门）→ [撤退判定 6.5]
  敌方回合：敌方AI行动 → [敌方逃跑判定 6.4]
  回合末：灼烧/灰蚀tick（6.3）→ 护盾清空 → 判定结局
} 直到 结局 ≠ null
战后.ts 汇总 → 返回 战斗结果
```
